<script>
(function () {
  "use strict";

  var currency = new Intl.NumberFormat('en-CA', { style: 'currency', currency: 'CAD', maximumFractionDigits: 0 });
  var TODAY = new Date(2026, 7, 12);
  var MS_DAY = 86400000;

  var state = { quotes: [] };

  var CUSTOMER_ALIASES = ['customer', 'client', 'job', 'description', 'job description', 'customer / job'];
  var AMOUNT_ALIASES = ['amount', 'quote', 'quote amount', 'price', 'total'];
  var DATE_ALIASES = ['date sent', 'quote date', 'sent date', 'date'];
  var CONTACT_ALIASES = ['last contact', 'last follow up', 'followed up', 'last contacted'];
  var STATUS_ALIASES = ['status', 'result', 'outcome'];

  function splitCSVLine(line) {
    var out = [], cur = '', q = false;
    for (var i = 0; i < line.length; i++) { var c = line[i]; if (c === '"') { q = !q; continue; } if (c === ',' && !q) { out.push(cur); cur = ''; continue; } cur += c; }
    out.push(cur); return out;
  }
  function findColumn(headers, aliases) {
    for (var a = 0; a < aliases.length; a++) { var idx = headers.indexOf(aliases[a]); if (idx !== -1) return idx; }
    for (var i = 0; i < headers.length; i++) for (var b = 0; b < aliases.length; b++) if (headers[i].indexOf(aliases[b]) !== -1) return i;
    return -1;
  }
  function parseDateFlexible(str) {
    var m = /^(\d{4})-(\d{1,2})-(\d{1,2})$/.exec(String(str).trim());
    if (m) return new Date(parseInt(m[1], 10), parseInt(m[2], 10) - 1, parseInt(m[3], 10));
    return new Date(str);
  }
  function normStatus(str) {
    var s = String(str || '').toLowerCase();
    if (s.indexOf('won') !== -1 || s.indexOf('yes') !== -1 || s.indexOf('close') !== -1) return 'won';
    if (s.indexOf('lost') !== -1 || s.indexOf('no') !== -1) return 'lost';
    return 'pending';
  }

  function parseCSV(text) {
    var lines = text.split(/\r?\n/).filter(function (l) { return l.trim().length; });
    if (lines.length < 2) return { error: 'That file has no rows to read.' };
    var headers = splitCSVLine(lines[0]).map(function (h) { return h.trim().toLowerCase(); });
    var custCol = findColumn(headers, CUSTOMER_ALIASES);
    var amtCol = findColumn(headers, AMOUNT_ALIASES);
    var dateCol = findColumn(headers, DATE_ALIASES);
    var contactCol = findColumn(headers, CONTACT_ALIASES);
    var statusCol = findColumn(headers, STATUS_ALIASES);
    if (amtCol === -1 || dateCol === -1) return { error: 'Couldn\'t find an amount and a date-sent column. Expected headers like "Customer", "Amount", "Date Sent", "Status".' };
    var quotes = [];
    for (var i = 1; i < lines.length; i++) {
      var f = splitCSVLine(lines[i]);
      if (!f[amtCol]) continue;
      var amount = parseFloat(String(f[amtCol]).replace(/[^0-9.\-]/g, ''));
      var sent = parseDateFlexible(f[dateCol]);
      if (isNaN(amount) || isNaN(sent.getTime())) continue;
      var contact = contactCol !== -1 && f[contactCol] ? parseDateFlexible(f[contactCol]) : null;
      if (contact && isNaN(contact.getTime())) contact = null;
      quotes.push({
        customer: custCol !== -1 ? (f[custCol] || 'Unlabelled').trim() : 'Unlabelled',
        amount: amount, sent: sent, contact: contact,
        status: statusCol !== -1 ? normStatus(f[statusCol]) : 'pending'
      });
    }
    if (!quotes.length) return { error: 'Found the columns, but no rows parsed cleanly. Check the date and amount formatting.' };
    return { quotes: quotes };
  }

  function daysBetween(a, b) { return Math.round((b - a) / MS_DAY); }

  function computeAll() {
    var q = state.quotes;
    var pending = q.filter(function (x) { return x.status === 'pending'; });
    var resolved = q.filter(function (x) { return x.status === 'won' || x.status === 'lost'; });

    pending.forEach(function (x) { x.daysSince = daysBetween(x.contact || x.sent, TODAY); });
    var pipelineValue = pending.reduce(function (s, x) { return s + x.amount; }, 0);
    var cold = pending.filter(function (x) { return x.daysSince >= 5; });
    var coldValue = cold.reduce(function (s, x) { return s + x.amount; }, 0);
    var needCall = pending.filter(function (x) { return x.daysSince >= 3; }).length;

    var fast = resolved.filter(function (x) { return x.contact && daysBetween(x.sent, x.contact) <= 2; });
    var slow = resolved.filter(function (x) { return !x.contact || daysBetween(x.sent, x.contact) > 2; });
    var fastWinPct = fast.length ? fast.filter(function (x) { return x.status === 'won'; }).length / fast.length : 0;
    var slowWinPct = slow.length ? slow.filter(function (x) { return x.status === 'won'; }).length / slow.length : 0;
    var overallWinPct = resolved.length ? resolved.filter(function (x) { return x.status === 'won'; }).length / resolved.length : 0;

    var sortedPending = pending.slice().sort(function (a, b) { return b.daysSince - a.daysSince; });

    return { pipelineValue: pipelineValue, coldValue: coldValue, coldCount: cold.length, needCall: needCall, fastWinPct: fastWinPct, slowWinPct: slowWinPct, overallWinPct: overallWinPct, sortedPending: sortedPending, fastCount: fast.length, slowCount: slow.length };
  }

  function urgencyClass(days) { return days >= 5 ? 'cold' : (days >= 3 ? 'soon' : 'ok'); }
  function urgencyLabel(days) { return days >= 5 ? days + 'd — going cold' : (days >= 3 ? days + 'd — follow up soon' : (days <= 0 ? 'sent today' : days + 'd — on track')); }

  function renderStats(agg) {
    document.getElementById('statPipeline').textContent = currency.format(agg.pipelineValue);
    document.getElementById('statCold').textContent = currency.format(agg.coldValue);
    document.getElementById('statColdSub').textContent = agg.coldCount + ' quote' + (agg.coldCount === 1 ? '' : 's') + ' untouched 5+ days';
    document.getElementById('statNeedCall').textContent = agg.needCall;
    document.getElementById('statCloseRate').textContent = (agg.overallWinPct * 100).toFixed(0) + '%';
    document.getElementById('heroFast').textContent = agg.fastCount ? (agg.fastWinPct * 100).toFixed(0) + '%' : '—';
    document.getElementById('heroSlow').textContent = agg.slowCount ? (agg.slowWinPct * 100).toFixed(0) + '%' : '—';
  }

  function renderBarChart(agg) {
    var svg = document.getElementById('barChart');
    var ns = 'http://www.w3.org/2000/svg';
    while (svg.firstChild) svg.removeChild(svg.firstChild);
    var W = 500, padL = 4, padR = 4, padT = 10, labelGap = 18, barH = 42, rowGap = 26;
    var rows = [
      { label: 'Followed up ≤ 2 days', pct: agg.fastWinPct, color: 'var(--series-1)', n: agg.fastCount },
      { label: 'Followed up later / never', pct: agg.slowWinPct, color: 'var(--series-2)', n: agg.slowCount }
    ];
    var maxW = W - padL - padR;
    var H = padT + rows.length * (labelGap + barH) + (rows.length - 1) * rowGap + 12;
    svg.setAttribute('viewBox', '0 0 ' + W + ' ' + H);
    rows.forEach(function (r, i) {
      var rowTop = padT + i * (labelGap + barH + rowGap);
      var barY = rowTop + labelGap;

      var lbl = document.createElementNS(ns, 'text');
      lbl.setAttribute('x', padL); lbl.setAttribute('y', rowTop + 10); lbl.setAttribute('class', 'axislabel');
      lbl.textContent = r.label;
      svg.appendChild(lbl);

      var track = document.createElementNS(ns, 'rect');
      track.setAttribute('x', padL); track.setAttribute('y', barY); track.setAttribute('width', maxW); track.setAttribute('height', barH); track.setAttribute('rx', 6);
      track.setAttribute('fill', 'var(--hairline-2)');
      svg.appendChild(track);

      var w = Math.max(2, maxW * r.pct);
      var fill = document.createElementNS(ns, 'rect');
      fill.setAttribute('x', padL); fill.setAttribute('y', barY); fill.setAttribute('width', w); fill.setAttribute('height', barH); fill.setAttribute('rx', 6);
      fill.setAttribute('fill', r.color);
      svg.appendChild(fill);

      var val = document.createElementNS(ns, 'text');
      val.setAttribute('y', barY + barH / 2 + 5); val.setAttribute('class', 'barlabel');
      if (w > maxW - 140) { val.setAttribute('x', padL + w - 10); val.setAttribute('text-anchor', 'end'); val.setAttribute('fill', 'white'); }
      else { val.setAttribute('x', padL + w + 10); }
      val.textContent = (r.pct * 100).toFixed(0) + '%  (' + r.n + ' quotes)';
      svg.appendChild(val);
    });
  }

  function renderPipelineTable(agg) {
    var t = document.getElementById('pipelineTable');
    if (!agg.sortedPending.length) { t.innerHTML = '<tbody><tr><td style="padding:0.8rem 0;color:var(--text-3);">Nothing open right now — pipeline is clear.</td></tr></tbody>'; return; }
    var html = '<thead><tr><th>Customer / job</th><th class="num">Amount</th><th>Sent</th><th>Status</th><th></th></tr></thead><tbody>';
    agg.sortedPending.forEach(function (x, i) {
      html += '<tr><td>' + x.customer + '</td><td class="num">' + currency.format(x.amount) + '</td><td>' + x.sent.toLocaleDateString('en-CA', { month: 'short', day: 'numeric' }) + '</td>' +
        '<td><span class="urgency ' + urgencyClass(x.daysSince) + '"><span class="dot"></span>' + urgencyLabel(x.daysSince) + '</span></td>' +
        '<td><button class="logbtn" data-log="' + i + '">Log follow-up</button></td></tr>';
    });
    html += '</tbody>';
    t.innerHTML = html;
    t.querySelectorAll('[data-log]').forEach(function (btn) {
      btn.addEventListener('click', function () {
        var idx = parseInt(btn.getAttribute('data-log'), 10);
        agg.sortedPending[idx].contact = new Date(TODAY);
        renderAll();
      });
    });
  }

  function renderInsights(agg) {
    var list = document.getElementById('insightList');
    list.innerHTML = '';
    var items = [];
    if (agg.fastCount && agg.slowCount) {
      var gap = agg.fastWinPct - agg.slowWinPct;
      items.push({ text: 'Quotes followed up within 2 days close at ' + (agg.fastWinPct * 100).toFixed(0) + '%. Left longer, that drops to ' + (agg.slowWinPct * 100).toFixed(0) + '% — same jobs, same prices, only the timing changed.', flag: gap > 0.15 });
    }
    if (agg.coldValue > 0) {
      items.push({ text: currency.format(agg.coldValue) + ' in open quotes hasn\'t been touched in 5+ days. Historically, most of that is already gone.', flag: true });
    }
    if (agg.needCall > 0) {
      items.push({ text: agg.needCall + ' open quote' + (agg.needCall === 1 ? '' : 's') + ' need' + (agg.needCall === 1 ? 's' : '') + ' a call today — see the table above, sorted by how long it\'s been.', flag: false });
    }
    items.push({ text: 'Overall close rate across the period is ' + (agg.overallWinPct * 100).toFixed(0) + '%. The gap between fast and slow follow-up is the cheapest lever available — it costs nothing to close more of what you already quoted.', flag: false });

    items.forEach(function (it) {
      var li = document.createElement('li'); if (it.flag) li.className = 'flag';
      var mark = document.createElement('span'); mark.className = 'mark';
      var span = document.createElement('span'); span.textContent = it.text;
      li.appendChild(mark); li.appendChild(span); list.appendChild(li);
    });
  }

  function renderAll() {

    renderStats(agg);
    renderBarChart(agg);
    renderPipelineTable(agg);
    renderInsights(agg);
  }

  function handleFile(file) {
    var reader = new FileReader();
    reader.onload = function () {
      var parsed = parseCSV(String(reader.result));
      var err = document.getElementById('parseError');

    reader.readAsText(file);
  }

  function mulberry32(seed) {
    return function () { seed |= 0; seed = (seed + 0x6D2B79F5) | 0; var t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; };
  }

  var CUSTOMERS = ['Sarah T.', 'Mike R.', 'The Nguyens', 'Dave P.', 'Colleen M.', 'Aaron W.', 'The Osei Family', 'Janet K.', 'Ryan B.', 'Priya S.', 'Tom H.', 'The Lindsays', 'Kevin O.', 'Marnie F.', 'Chris D.', 'Alicia G.', 'Brad N.', 'The Fenwicks', 'Wendy C.', 'Steve M.'];

  function generateSample() {
    var rand = mulberry32(20260812);
    var quotes = [];
    var startDate = new Date(2026, 1, 15); // Feb 15
    var totalDays = daysBetween(startDate, TODAY);
    var count = 150;
    for (var i = 0; i < count; i++) {
      var jt = JOB_TYPES[Math.floor(rand() * JOB_TYPES.length)];
      var amount = Math.round(jt.lo + rand() * (jt.hi - jt.lo));
      var sentOffset = Math.floor(rand() * totalDays);
      var sent = new Date(startDate.getTime() + sentOffset * MS_DAY);
      var customer = CUSTOMERS[Math.floor(rand() * CUSTOMERS.length)] + ' — ' + jt.name;
      var ageDays = daysBetween(sent, TODAY);

      var fastFollowUp = rand() < 0.55;
      var followUpDelay = fastFollowUp ? Math.floor(rand() * 3) : (3 + Math.floor(rand() * 8));

      if (ageDays <= 12) {
        var hasContact = ageDays > 1 ? rand() < 0.5 : false;
        var contact = hasContact ? new Date(sent.getTime() + Math.min(followUpDelay, ageDays) * MS_DAY) : null;
        quotes.push({ customer: customer, amount: amount, sent: sent, contact: contact, status: 'pending' });
      } else {
        var contactDate = new Date(sent.getTime() + Math.min(followUpDelay, ageDays - 1) * MS_DAY);
        var winProb = fastFollowUp ? 0.60 : 0.24;
        var won = rand() < winProb;
        quotes.push({ customer: customer, amount: amount, sent: sent, contact: contactDate, status: won ? 'won' : 'lost' });
      }
    }
    return quotes;
  }
